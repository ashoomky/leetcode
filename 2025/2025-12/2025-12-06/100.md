this is a way comparing the equality of two lists.
```python
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution(object):
    def isSameTree(self, p, q):
        """
        :type p: Optional[TreeNode]
        :type q: Optional[TreeNode]
        :rtype: bool
        """
        self.tree_one = []
        self.tree_two = []

        def dfs(root, tree_list):
            if not root:
                tree_list.append(None)
                return 0
                
            tree_list.append(root.val)
            dfs(root.left, tree_list)
            dfs(root.right, tree_list)
           
            return 0
        
        dfs(p, self.tree_one)
        dfs(q, self.tree_two)
        return self.tree_one == self.tree_two
```

by comparing the actual subtrees of left and right we can get the correct result.
```python
# Definition for a binary tree node.
# class TreeNode(object):
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution(object):
    def isSameTree(self, p, q):
        """
        :type p: Optional[TreeNode]
        :type q: Optional[TreeNode]
        :rtype: bool
        """
        if not p and not q:
            return True
        
        if p and q and p.val == q.val:
            left = self.isSameTree(p.left, q.left) 
            right = self.isSameTree(p.right, q.right)
            return left and right
        else:
            return False
```
