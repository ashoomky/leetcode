okay so a doubly-linked list is to bidirectionally travel in a linked list, for efficient insertions / deletions at any point in a sequence. 

```python
class Node():
    def __init__(self, key, val):
        self.key = key
        self.val = val
        self.prev = self.next = None


class LRUCache(object):

    def __init__(self, capacity):
        """
        :type capacity: int
        """
        self.capacity = capacity
        self.cache = {} # maps key to node

        # keeping track of the least and most recently used nodes using dummy nodes. left is least, right is most
        self.left, self.right = Node(0, 0), Node(0, 0)
        # want them to be linked to each other initially
        self.left.next, self.right.prev = self.right, self.left

    #remove node from linked list
    def remove(self, node):
        prev, nxt = node.prev, node.next
        prev.next, nxt.prev = nxt, prev
    
    # insert at the most used side (right)
    def insert(self, node):
        prev, nxt = self.right.prev, self.right
        prev.next = nxt.prev = node
        node.prev = prev
        node.next = nxt

    def get(self, key):
        """
        :type key: int
        :rtype: int
        """
        if key in self.cache:
            #return the value because the key is mapped to a node object
            self.remove(self.cache[key])
            self.insert(self.cache[key])
            return self.cache[key].val
        return -1

    def put(self, key, value):
        """
        :type key: int
        :type value: int
        :rtype: None
        """
        if key in self.cache:
            self.remove(self.cache[key])
        self.cache[key] = Node(key, value)
        # insert node into linked list 
        self.insert(self.cache[key])

        if len(self.cache) > self.capacity:
            #remove LRU from linked list and delete from cache
            lru = self.left.next
            self.remove(lru)
            del self.cache[lru.key]
```
