class Solution(object):
    def twoSum(self, numbers, target):
        """
        :type numbers: List[int]
        :type target: int
        :rtype: List[int]
        """
        ptr_1 = 0
        for i in range(len(numbers)):
            ptr_2 = len(numbers) - 1
            while ptr_1 != ptr_2:
                if (numbers[ptr_1] + numbers[ptr_2]) > target or (numbers[ptr_1] + numbers[ptr_2]) != target:
                    ptr_2 -= 1
                elif (numbers[ptr_1] + numbers[ptr_2]) == target:
                    return [ptr_1+1, ptr_2+1]
            ptr_1 += 1
        return []

```python
class Solution(object):
    def twoSum(self, numbers, target):
        """
        :type numbers: List[int]
        :type target: int
        :rtype: List[int]
        """
        l, r = 0, len(numbers) - 1

        while l < r:
            sum = numbers[l] + numbers[r]
            if sum > target:
                r -= 1
            elif sum < target:
                l += 1
            else:
                return [l + 1, r + 1]
        return []
```
